# Phase 2: Quick Look Extension UX Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Make the Quick Look preview surface a respectable standalone product. Add code-block copy affordance, ⌘+/⌘−/⌘0 zoom, atomic-save reload, native find bar, robust link click handling, and inline image polish.

**Architecture:** All work lives in the `MarkdownQuickLookPreviewExtension` target. We add an overlay layer on top of the existing `MarkdownTextView` (NSViewRepresentable wrapper around `NSTextView`) for the copy-button affordance, intercept key events for zoom, attach an `NSFilePresenter` for save reload, and configure native `NSTextView` finder/link handling. No host-app changes here — that's Phase 5.

**Tech Stack:** Swift, AppKit (`NSTextView`, `NSFindBarContainer`, `NSFilePresenter`, `NSTrackingArea`, `NSPasteboard`, `NSWorkspace`), SwiftUI (`NSViewRepresentable`), `DispatchSource`, App Group `UserDefaults`.

**Spec:** `docs/superpowers/specs/2026-05-23-pluk-gap-roadmap-design.md` § Phase 2.

### Task 4: Atomic-save reload (auto-refresh on external edits)

The QL preview is short-lived but visible while spacebar is held. If an editor saves the file (often by writing a temp file + renaming), the preview goes stale. We watch the original file via `DispatchSource.makeFileSystemObjectSource` and trigger a re-render. For atomic save we have to re-open by path (the inode changes).

**Files:**
- Create: `MarkdownQuickLookPreviewExtension/FileWatcher.swift`
- Modify: `MarkdownQuickLookPreviewExtension/PreviewViewController.swift`
- Create: `MarkdownQuickLookPreviewExtension/Tests/FileWatcherTests.swift`

- [ ] **Step 1: Define the watcher**

Create `MarkdownQuickLookPreviewExtension/FileWatcher.swift`:

```swift
import Foundation

/// Watches a file path for any change (write, rename, delete). Re-opens the
/// path on each event so atomic-save patterns (write tmp + rename) trigger
/// the same notification stream.
final class FileWatcher {
    typealias Handler = @Sendable () -> Void

    private let url: URL
    private let onChange: Handler
    private let queue: DispatchQueue
    private var source: DispatchSourceFileSystemObject?
    private var fd: Int32 = -1

    init(url: URL, queue: DispatchQueue = .main, onChange: @escaping Handler) {
        self.url = url
        self.queue = queue
        self.onChange = onChange
    }

    func start() {
        guard source == nil else { return }
        let descriptor = open(url.path, O_EVTONLY)
        guard descriptor >= 0 else { return }
        fd = descriptor

        let src = DispatchSource.makeFileSystemObjectSource(
            fileDescriptor: descriptor,
            eventMask: [.write, .extend, .attrib, .link, .rename, .delete],
            queue: queue
        )
        let handler = onChange
        src.setEventHandler { handler() }
        src.setCancelHandler { [fd] in
            if fd >= 0 { close(fd) }
        }
        src.resume()
        source = src
    }

    func stop() {
        source?.cancel()
        source = nil
        fd = -1
    }

    deinit { stop() }
}
```

- [ ] **Step 4: Build the overlay view**

Create `MarkdownQuickLookPreviewExtension/CodeBlockCopyOverlay.swift`:

```swift
import AppKit

/// Sits over the NSTextView. For each code block, hover within its rect shows
/// a "Copy" button in the top-right of the block.
final class CodeBlockCopyOverlay: NSView {
    struct Region {
        let range: NSRange
        let code: String
    }

    private weak var textView: NSTextView?
    private var regions: [Region] = []
    private var trackingArea: NSTrackingArea?
    private var activeButton: NSButton?
    private var activeRegionIndex: Int?

    init(textView: NSTextView) {
        self.textView = textView
        super.init(frame: .zero)
        self.wantsLayer = true
        self.layer?.backgroundColor = .clear
    }
    required init?(coder: NSCoder) { fatalError("init(coder:) not implemented") }

    override var isFlipped: Bool { true }

    func update(regions: [Region]) {
        self.regions = regions
        activeButton?.removeFromSuperview()
        activeButton = nil
        activeRegionIndex = nil
        rebuildTrackingArea()
    }

    override func updateTrackingAreas() {
        super.updateTrackingAreas()
        rebuildTrackingArea()
    }

    private func rebuildTrackingArea() {
        if let ta = trackingArea { removeTrackingArea(ta) }
        let ta = NSTrackingArea(
            rect: bounds,
            options: [.activeInKeyWindow, .mouseMoved, .inVisibleRect],
            owner: self,
            userInfo: nil
        )
        addTrackingArea(ta)
        trackingArea = ta
    }

    override func mouseMoved(with event: NSEvent) {
        guard let textView,
              let layoutManager = textView.layoutManager,
              let textContainer = textView.textContainer else { return }
        let local = convert(event.locationInWindow, from: nil)

        for (i, region) in regions.enumerated() {
            let glyphRange = layoutManager.glyphRange(forCharacterRange: region.range, actualCharacterRange: nil)
            let rect = layoutManager.boundingRect(forGlyphRange: glyphRange, in: textContainer)
            let rectInOverlay = textView.convert(rect, to: self)
            if rectInOverlay.contains(local) {
                showButton(for: i, in: rectInOverlay)
                return
            }
        }
        hideButton()
    }

    private func showButton(for index: Int, in rect: NSRect) {
        if activeRegionIndex == index { return }
        hideButton()

        let button = NSButton(title: "Copy", target: self, action: #selector(handleCopy(_:)))
        button.bezelStyle = .rounded
        button.controlSize = .small
        button.tag = index
        button.translatesAutoresizingMaskIntoConstraints = false
        addSubview(button)

        NSLayoutConstraint.activate([
            button.topAnchor.constraint(equalTo: topAnchor, constant: rect.minY + 6),
            button.trailingAnchor.constraint(equalTo: leadingAnchor, constant: rect.maxX - 6)
        ])

        activeButton = button
        activeRegionIndex = index
    }

    private func hideButton() {
        activeButton?.removeFromSuperview()
        activeButton = nil
        activeRegionIndex = nil
    }

    @objc private func handleCopy(_ sender: NSButton) {
        let index = sender.tag
        guard index >= 0 && index < regions.count else { return }
        let code = regions[index].code
        let pb = NSPasteboard.general
        pb.clearContents()
        pb.setString(code, forType: .string)

        sender.title = "Copied"
        DispatchQueue.main.asyncAfter(deadline: .now() + 1.0) { [weak sender] in
            sender?.title = "Copy"
        }
    }
}
```

- [ ] **Step 2: Add a minimal `<img>` parser**

```swift
private static let imgTagRegex: NSRegularExpression = {
    try! NSRegularExpression(
        pattern: #"^<img\s+([^>]*)>$"#,
        options: [.caseInsensitive]
    )
}()
private static let imgSrcRegex: NSRegularExpression = {
    try! NSRegularExpression(pattern: #"src\s*=\s*"([^"]+)""#, options: [.caseInsensitive])
}()
private static let imgWidthRegex: NSRegularExpression = {
    try! NSRegularExpression(pattern: #"width\s*=\s*"?(\d+)"?"#, options: [.caseInsensitive])
}()
private static let imgAltRegex: NSRegularExpression = {
    try! NSRegularExpression(pattern: #"alt\s*=\s*"([^"]*)""#, options: [.caseInsensitive])
}()

private func rawImageTag(from line: String) -> (alt: String, src: String, maxWidth: CGFloat?)? {
    let trimmed = line.trimmingCharacters(in: .whitespaces)
    let ns = trimmed as NSString
    let range = NSRange(location: 0, length: ns.length)
    guard Self.imgTagRegex.firstMatch(in: trimmed, options: .anchored, range: range) != nil else { return nil }

    func capture(_ regex: NSRegularExpression) -> String? {
        guard let m = regex.firstMatch(in: trimmed, range: range) else { return nil }
        return ns.substring(with: m.range(at: 1))
    }
    guard let src = capture(Self.imgSrcRegex) else { return nil }
    let alt = capture(Self.imgAltRegex) ?? ""
    let width = capture(Self.imgWidthRegex).flatMap { Double($0) }.map { CGFloat($0) }
    return (alt, src, width)
}
```
